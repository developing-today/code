# The empty string and a lone NUL.
tuple	s: s:\0	ok
tuple	s:\0 s:\0\0	ok
# Values that differ only after a shared prefix.
tuple	s:ab s:a\0b	ok
tuple	s:a\x01 s:a	ok
tuple	s:\xc3\xa9 s:e	ok
# The same cases for byte strings.
tuple	b:a b:a\0b	ok
tuple	b:\0 b:\0\0	ok
tuple	b:a\0 b:a\0\0	ok
tuple	b:\0\0 b:\0	ok
tuple	b:\xff b:\xff\0	ok
# Mixed element types: bytes sort before strings.
tuple	b:a s:a	ok
